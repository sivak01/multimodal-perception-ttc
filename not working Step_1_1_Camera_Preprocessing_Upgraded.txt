    "# ─────────────────────────────────────────────────────────────────\n",
    "# CELL 2 — Main pipeline: extract camera metadata for all samples\n",
    "#\n",
    "# For each sample, reads sensor_meta.json (from Step 0) and the\n",
    "# nuScenes API to extract per-camera:\n",
    "#   - filename, timestamp, ego_pose\n",
    "#   - sensor_to_ego translation + rotation (extrinsics)\n",
    "#   - camera_intrinsic (3x3 matrix)\n",
    "#\n",
    "# FIXED: removed accidental comment injection inside dict literal\n",
    "# (original had a comment string mixed into the \"rotation\" key line)\n",
    "# ─────────────────────────────────────────────────────────────────\n",
    "\n",
    "import json\n",
    "from pathlib import Path\n",
    "from nuscenes.nuscenes import NuScenes\n",
    "from config import DATA_ROOT, NUSCENES_VERSION, STEP0_DIR, STEP1_DIR\n",
    "\n",
    "# ── Load nuScenes (for intrinsic lookup only) ──────────────────────\n",
    "nusc = NuScenes(\n",
    "    version=NUSCENES_VERSION,\n",
    "    dataroot=str(DATA_ROOT),\n",
    "    verbose=False\n",
    ")\n",
    "\n",
    "# ── Load Step 0 index ──────────────────────────────────────────────\n",
    "with open(STEP0_DIR / \"samples_index.json\") as f:\n",
    "    samples_index = json.load(f)\n",
    "\n",
    "camera_data = {}\n",
    "\n",
    "# ── Process each sample ────────────────────────────────────────────\n",
    "for sample_id, sample_info in samples_index.items():\n",
    "\n",
    "    # Reconstruct folder path from config + relative folder name\n",
    "    sample_folder     = STEP0_DIR / sample_info[\"folder\"]\n",
    "    sensor_meta_path  = sample_folder / \"sensor_meta.json\"\n",
    "\n",
    "    with open(sensor_meta_path) as f:\n",
    "        sensor_meta = json.load(f)\n",
    "\n",
    "    cam_views = {}\n",
    "\n",
    "    for channel, meta in sensor_meta[\"sensor_channels\"].items():\n",
    "        if meta[\"modality\"] != \"camera\":\n",
    "            continue\n",
    "\n",
    "        # Get camera_intrinsic from nuScenes API\n",
    "        cs = nusc.get(\"calibrated_sensor\", meta[\"calibrated_sensor_token\"])\n",
    "\n",
    "        cam_views[channel] = {\n",
    "            \"filename\"  : meta[\"filename\"],\n",
    "            \"timestamp\" : meta[\"timestamp\"],\n",
    "            \"ego_pose\"  : meta[\"ego_pose\"],\n",
    "            \"calibrated_sensor\": {\n",
    "                \"translation\"     : meta[\"sensor_to_ego_translation\"],\n",
    "                \"rotation\"        : meta[\"sensor_to_ego_rotation\"],   # ← FIXED: was corrupted\n",
    "                \"camera_intrinsic\": cs[\"camera_intrinsic\"]\n",
    "            }\n",
    "        }\n",
    "\n",
    "    if cam_views:\n",
    "        camera_data[sample_id] = cam_views\n",
    "\n",
    "# ── Save output ────────────────────────────────────────────────────\n",
    "output_file = STEP1_DIR / \"camera_meta.json\"\n",
    "with open(output_file, \"w\") as f:\n",
    "    json.dump(camera_data, f, indent=2)\n",
    "\n",
    "# ── Validate ───────────────────────────────────────────────────────\n",
    "assert len(camera_data) == len(samples_index), \\\n",
    "    f\"Expected {len(samples_index)} samples, got {len(camera_data)}\"\n",
    "\n",
    "print(f\"✅ Step 1.1 complete: {len(camera_data)} samples processed.\")\n",
    "print(f\"📄 Saved to: {output_file}\")"
   ]
